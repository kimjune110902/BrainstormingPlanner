  function linkPath(a, b, fromPort, toPort) {
    const dx = Math.abs(b.x - a.x);
    const dy = Math.abs(b.y - a.y);
    const strength = Math.max(44, Math.min(180, (dx + dy) * .28));
    const c1 = { x: a.x, y: a.y };
    const c2 = { x: b.x, y: b.y };
    if (fromPort === "right") c1.x += strength;
    if (fromPort === "left") c1.x -= strength;
    if (fromPort === "bottom") c1.y += strength;
    if (fromPort === "top") c1.y -= strength;
    if (toPort === "right") c2.x += strength;
    if (toPort === "left") c2.x -= strength;
    if (toPort === "bottom") c2.y += strength;
    if (toPort === "top") c2.y -= strength;
    return `M ${a.x} ${a.y} C ${c1.x} ${c1.y}, ${c2.x} ${c2.y}, ${b.x} ${b.y}`;
  }

  function drawLinks() {
    els.connectionLayer.innerHTML = "";
    state.links.forEach((link) => {
      const fromNode = getNode(link.from);
      const toNode = getNode(link.to);
      if (!fromNode || !toNode) return;
      if (state.filterType !== "all" && (fromNode.type !== state.filterType || toNode.type !== state.filterType)) return;
      
      const color = colors[fromNode.color] || colors.plum;
      // Using World Coordinates directly inside the scaled world layer
      const a = portPosition(link.from, link.fromPort);
      const b = portPosition(link.to, link.toPort);
      
      const path = document.createElementNS("http://www.w3.org/2000/svg", "path");
      path.setAttribute("d", linkPath(a, b, link.fromPort, link.toPort));
      path.setAttribute("fill", "none");
      path.setAttribute("stroke", color.stroke);
      path.setAttribute("stroke-width", "2.2");
      path.setAttribute("stroke-linecap", "round");
      path.setAttribute("opacity", ".78");
      path.setAttribute("marker-end", "url(#arrow)");
      els.connectionLayer.appendChild(path);
      
      // Increased touch target size for mobile link clearing
      const hit = document.createElementNS("http://www.w3.org/2000/svg", "path");
      hit.setAttribute("d", path.getAttribute("d"));
      hit.setAttribute("fill", "none");
      hit.setAttribute("stroke", "transparent");
      hit.setAttribute("stroke-width", "24"); 
      hit.style.pointerEvents = "stroke";
      hit.style.cursor = "pointer";
      
      const clearLink = (e) => {
        e.stopPropagation();
        state.links = state.links.filter((item) => item.id !== link.id);
        drawLinks();
        saveSoon();
        toast("Connection removed");
      };
      hit.addEventListener("pointerdown", clearLink);
      els.connectionLayer.appendChild(hit);
    });
  }

  function zoom(delta) {
    const old = state.scale;
    const next = clamp(old + delta, .24, 2.5);
    const rect = els.canvasShell.getBoundingClientRect();
    const cx = rect.width / 2;
    const cy = rect.height / 2;
    state.pan.x = cx - (cx - state.pan.x) * (next / old);
    state.pan.y = cy - (cy - state.pan.y) * (next / old);
    state.scale = next;
    applyTransform();
    drawLinks();
    saveSoon();
  }

  function fitView() {
    if (!state.nodes.length) return;
    const rect = els.canvasShell.getBoundingClientRect();
    const xs = state.nodes.map((node) => [node.x, node.x + node.w]).flat();
    const ys = state.nodes.map((node) => [node.y, node.y + node.h]).flat();
    const minX = Math.min(...xs);
    const maxX = Math.max(...xs);
    const minY = Math.min(...ys);
    const maxY = Math.max(...ys);
    const pad = 80;
    const width = maxX - minX + pad * 2;
    const height = maxY - minY + pad * 2;
    state.scale = clamp(Math.min(rect.width / width, rect.height / height), .32, 1.8);
    state.pan.x = (rect.width - (maxX - minX) * state.scale) / 2 - minX * state.scale;
    state.pan.y = (rect.height - (maxY - minY) * state.scale) / 2 - minY * state.scale;
    applyTransform();
    drawLinks();
    saveSoon();
  }

    if (from === to) return;

  }

  function sheetBounds() {

  function wrapSvgText(text, x, y, maxChars, lineHeight, cls) {
    const words = String(text || "").split(/\s+/).filter(Boolean);
    const lines = [];
    let line = "";
    words.forEach((word) => {
      const next = line ? `${line} ${word}` : word;
      if (next.length > maxChars && line) {
        lines.push(line);
        line = word;
      } else {
        line = next;
      }
    });
    if (line) lines.push(line);
    return lines.slice(0, 5).map((value, index) => `<text class="${cls}" x="${x}" y="${y + index * lineHeight}">${escapeHtml(value)}</text>`).join("");
  }

  function buildSnapshotSvg(bounds) {
    const links = state.links.map((link) => {
      const from = getNode(link.from);
      const to = getNode(link.to);
      if (!from || !to) return "";
      const color = colors[from.color] || colors.plum;
      const a = portPosition(link.from, link.fromPort);
      const b = portPosition(link.to, link.toPort);
      const path = linkPath({ x: a.x - bounds.x, y: a.y - bounds.y }, { x: b.x - bounds.x, y: b.y - bounds.y }, link.fromPort, link.toPort);
      return `<path d="${path}" fill="none" stroke="${color.stroke}" stroke-width="2.4" stroke-linecap="round" marker-end="url(#arrowSnap)" opacity=".72"/>`;
    }).join("");
    const nodes = state.nodes.map((node) => {
      const color = colors[node.color] || colors.plum;
      const radius = node.shape === "sharp" ? 4 : node.shape === "pill" ? 22 : 8;
      const x = node.x - bounds.x;
      const y = node.y - bounds.y;
      const maxChars = Math.max(12, Math.floor((node.w - 28) / 7));
      return `
        <g>
          <rect x="${x}" y="${y}" width="${node.w}" height="${node.h}" rx="${radius}" fill="${color.bg}" stroke="${color.line}" stroke-width="${node.type === "main" ? 2.2 : 1.6}"/>
          ${wrapSvgText(node.title, x + 14, y + 25, maxChars, 16, "title")}
          ${wrapSvgText(node.note, x + 14, y + 50, maxChars + 5, 15, "note")}
        </g>
      `;
    }).join("");
    return `<svg xmlns="http://www.w3.org/2000/svg" width="${Math.ceil(bounds.w)}" height="${Math.ceil(bounds.h)}" viewBox="0 0 ${Math.ceil(bounds.w)} ${Math.ceil(bounds.h)}">
      <defs><marker id="arrowSnap" markerWidth="9" markerHeight="9" refX="7" refY="4.5" orient="auto"><path d="M1 1 L8 4.5 L1 8 Z" fill="#7256ad" opacity=".62"/></marker></defs>
      <style>.title{font:700 14px system-ui,sans-serif;fill:#211b24}.note{font:12px system-ui,sans-serif;fill:#6f6574}</style>
      <rect width="100%" height="100%" fill="#fbfaf7"/>
      ${links}
      ${nodes}
    </svg>`;
  }

  function importFile(file) {
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const parsed = JSON.parse(reader.result);
        const saved = parsed.state || parsed;
        state.nodes = Array.isArray(saved.nodes) ? saved.nodes : [];
        state.links = Array.isArray(saved.links) ? saved.links : [];
        state.tasks = Array.isArray(saved.tasks) ? saved.tasks : [];
        state.nextNode = saved.nextNode || state.nodes.length + 1;
        state.nextLink = saved.nextLink || state.links.length + 1;
        state.nextTask = saved.nextTask || state.tasks.length + 1;
        state.pan = saved.pan || state.pan;
        state.scale = saved.scale || state.scale;
        state.grid = saved.grid !== false;
        saveNow();
        renderAll();
        renderPlanner();
        toast("Sheet imported");
      } catch {
        toast("Could not import this file");
      }
    };
    reader.readAsText(file);
  }

        isCustomSize: false // Automatically sets proportional sizing relative to text inputs
      });
      els.nodeWizardModal.classList.remove("show");
    });

    document.getElementById("selectTool").addEventListener("click", () => setTool("select"));
    document.getElementById("connectTool").addEventListener("click", () => setTool("connect"));
    document.getElementById("fitBtn").addEventListener("click", fitView);
    document.getElementById("zoomInBtn").addEventListener("click", () => zoom(.12));
    document.getElementById("zoomOutBtn").addEventListener("click", () => zoom(-.12));
    
    document.getElementById("downloadBtn").addEventListener("click", (e) => {
      e.stopPropagation();
      els.downloadMenu.classList.toggle("show");
    });
    document.getElementById("downloadSheetBtn").addEventListener("click", () => {
      els.downloadMenu.classList.remove("show");
      downloadJson();
    });
    document.getElementById("downloadImageBtn").addEventListener("click", () => {
      els.downloadMenu.classList.remove("show");
      downloadImage();
    });
    document.getElementById("importBtn").addEventListener("click", () => els.fileInput.click());
    els.fileInput.addEventListener("change", () => {
      if (els.fileInput.files[0]) importFile(els.fileInput.files[0]);
      els.fileInput.value = "";
    });

    // Close on mobile back/toggle action
    document.getElementById("sheetToggle").addEventListener("click", (e) => {
      e.stopPropagation();
      closeInspector();
    });

    els.canvasWorld.addEventListener("pointerdown", (e) => {
      const nodeEl = e.target.closest(".node");
      if (!nodeEl) return;
      const node = getNode(nodeEl.dataset.id);
      if (!node) return;
      
      if (e.target.closest(".port")) {
        if (state.tool !== "connect") setTool("connect");
        startDraft(node.id, e.target.closest(".port").dataset.port, e);
        return;
      }
      
      // Auto-fallback mechanics from Connect to Move Mode when clicking nodes
      if (state.tool === "connect") {
        setTool("select");
        selectNode(node.id);
        return;
      }

      if (e.target.closest(".resize-corner")) {
        node.isCustomSize = true; // Enforces tracking standard locks if manual adjust triggers
        resize = { node, startX: e.clientX, startY: e.clientY, w: node.w, h: node.h };
        nodeEl.setPointerCapture(e.pointerId);
        e.preventDefault();
        return;
      }

      selectNode(node.id);
      node.z = ++state.nextNode;
      const point = pointerWorld(e);
      drag = { node, offsetX: point.x - node.x, offsetY: point.y - node.y, pointerId: e.pointerId };
      nodeEl.setPointerCapture(e.pointerId);
      e.preventDefault();
    });

    els.canvasShell.addEventListener("pointerdown", (e) => {
      if (e.target.closest(".node") || e.target.closest(".toolrail") || e.target.closest(".floating-bar") || e.target.closest(".modal-shade")) return;
      
      // Dynamic fallback mechanics when clicking open areas
      if (state.tool === "connect") {
        setTool("select");
        return;
      }
      
      if (state.tool === "select") {
        panDrag = { startX: e.clientX, startY: e.clientY, x: state.pan.x, y: state.pan.y, pointerId: e.pointerId };
        els.canvasShell.setPointerCapture(e.pointerId);
      }
      selectNode(null);
    });

    window.addEventListener("pointermove", (e) => {
      if (drag) {
        const point = pointerWorld(e);
        drag.node.x = point.x - drag.offsetX;
        drag.node.y = point.y - drag.offsetY;
        renderNodes();
        drawLinks();
      } else if (resize) {
        // Enforcing safety clamps on minimum proportions to stop flooding bugs
        resize.node.w = clamp(resize.w + (e.clientX - resize.startX) / state.scale, 140, 480);
        resize.node.h = clamp(resize.h + (e.clientY - resize.startY) / state.scale, 80, 400);
        renderNodes();
        drawLinks();
        renderInspector();
      } else if (panDrag) {
        state.pan.x = panDrag.x + e.clientX - panDrag.startX;
        state.pan.y = panDrag.y + e.clientY - panDrag.startY;
        applyTransform();
      } else if (connectDraft) {
        const start = portPosition(connectDraft.nodeId, connectDraft.port);
        const point = pointerWorld(e);
        els.draftConnection.setAttribute("d", `M ${start.x} ${start.y} L ${point.x} ${point.y}`);
      }
    }, { passive: false });

    window.addEventListener("pointerup", (e) => {
      if (drag || resize || panDrag) saveSoon();
      drag = null;
      resize = null;
      panDrag = null;
      if (connectDraft) {
        const target = document.elementFromPoint(e.clientX, e.clientY)?.closest(".port");
        if (target) {
          const nodeEl = target.closest(".node");
          addConnection(connectDraft.nodeId, connectDraft.port, nodeEl.dataset.id, target.dataset.port);
        }
        connectDraft = null;
        els.draftConnection.style.display = "none";
        document.querySelectorAll(".port").forEach((el) => el.classList.remove("armed"));
        setTool("select"); // Drops straight into select layout once path closes
      }
    });

    els.canvasShell.addEventListener("wheel", (e) => {
      e.preventDefault();
      zoom(e.deltaY > 0 ? -.08 : .08);
    }, { passive: false });

    els.inspectorBody.addEventListener("input", (e) => {
      const node = selectedNode();
      if (!node) return;
      if (e.target.id === "nodeTitle") {
        node.title = e.target.value;
        renderNodes();
        drawLinks();
        saveSoon();
      }
      if (e.target.id === "nodeNote") {
        node.note = e.target.value;
        renderNodes();
        drawLinks();
        saveSoon();
      }
      if (e.target.id === "nodeWidth" || e.target.id === "nodeWidthText") {
        node.w = clamp(Number(e.target.value) || node.w, 140, 480);
        node.isCustomSize = true;
        const range = document.getElementById("nodeWidth");
        const text = document.getElementById("nodeWidthText");
        if (range && range !== e.target) range.value = node.w;
        if (text && text !== e.target) text.value = node.w;
        renderNodes();
        drawLinks();
        saveSoon();
      }
    });

    els.inspectorBody.addEventListener("click", (e) => {
      const type = e.target.closest("[data-node-type]");
      if (type) updateSelected({ type: type.dataset.nodeType });
      const shape = e.target.closest("[data-node-shape]");
      if (shape) updateSelected({ shape: shape.dataset.nodeShape });
      const color = e.target.closest("[data-node-color]");
      if (color) updateSelected({ color: color.dataset.nodeColor });
      
      if (e.target.closest("#duplicateNodeBtn")) duplicateSelected();
      if (e.target.closest("#deleteNodeBtn")) deleteSelected();
      if (e.target.closest("#bringFrontBtn")) {
        const node = selectedNode();
        if (node) updateSelected({ z: ++state.nextNode });
      }
      if (e.target.closest("#makeTaskBtn")) {
        const node = selectedNode();
        if (node) openTaskModal(isoDate(new Date()), node);
      }
    });

    document.querySelectorAll("[data-type-filter]").forEach((button) => {
      button.addEventListener("click", () => {
        state.filterType = button.dataset.typeFilter;
        document.querySelectorAll("[data-type-filter]").forEach((item) => item.classList.toggle("active", item === button));
        renderAll();
      });
    });

    document.getElementById("prevMonthBtn").addEventListener("click", () => {
      state.month = new Date(state.month.getFullYear(), state.month.getMonth() - 1, 1);
      renderPlanner();
    });
    document.getElementById("nextMonthBtn").addEventListener("click", () => {
      state.month = new Date(state.month.getFullYear(), state.month.getMonth() + 1, 1);
      renderPlanner();
    });
    document.getElementById("addTaskTopBtn").addEventListener("click", () => openTaskModal());
    document.querySelectorAll("[data-plan-view]").forEach((button) => button.addEventListener("click", () => {
      state.plannerView = button.dataset.planView;
      renderPlanner();
    }));

    document.getElementById("tagFilters").addEventListener("click", (e) => {
      const button = e.target.closest("[data-tag]");
      if (!button) return;
      state.activeTag = button.dataset.tag;
      document.querySelectorAll("[data-tag]").forEach((item) => item.classList.toggle("active", item === button));
      renderPlanner();
    });

    els.plannerScroll.addEventListener("click", (e) => {
      const addDate = e.target.closest("[data-add-task-date]");
      if (addDate) openTaskModal(addDate.dataset.addTaskDate);
      const toggle = e.target.closest("[data-toggle-task]");
      if (toggle) {
        const task = state.tasks.find((item) => item.id === toggle.dataset.toggleTask);
        if (task) task.done = !task.done;
        renderPlanner();
        saveSoon();
      }
      const del = e.target.closest("[data-delete-task]");
      if (del) {
        state.tasks = state.tasks.filter((item) => item.id !== del.dataset.deleteTask);
        renderPlanner();
        saveSoon();
      }
    });

    els.taskColorChoices.addEventListener("click", (e) => {
      const button = e.target.closest("[data-task-color]");
      if (!button) return;
      taskColor = button.dataset.taskColor;
      renderTaskColors();
    });

    document.getElementById("cancelTaskBtn").addEventListener("click", closeTaskModal);
    document.getElementById("saveTaskBtn").addEventListener("click", saveTask);
    
    // Background light tap drops modal shades
    els.taskModal.addEventListener("click", (e) => {
      if (e.target === els.taskModal) closeTaskModal();
    });
    els.nodeWizardModal.addEventListener("click", (e) => {
      if (e.target === els.nodeWizardModal) els.nodeWizardModal.classList.remove("show");
    });

    els.taskName.addEventListener("keydown", (e) => {
      if (e.key === "Enter") saveTask();
    });

    window.addEventListener("keydown", (e) => {
      if (["INPUT", "TEXTAREA", "SELECT"].includes(document.activeElement.tagName)) return;
      if (e.key === "Escape") {
        setTool("select");
        closeTaskModal();
        closeInspector();
        els.nodeWizardModal.classList.remove("show");
      }
      if ((e.key === "Delete" || e.key === "Backspace") && state.selectedNode) deleteSelected();
      if (e.key.toLowerCase() === "n" && state.view === "brainstorm") openWizard();
      if (e.key.toLowerCase() === "c" && state.view === "brainstorm") setTool("connect");
      if (e.key.toLowerCase() === "s" && state.view === "brainstorm") setTool("select");
      if (e.key.toLowerCase() === "f" && state.view === "brainstorm") fitView();
    });

    document.addEventListener("click", (e) => {
      if (!e.target.closest("#downloadMenu") && !e.target.closest("#downloadBtn")) {
        els.downloadMenu.classList.remove("show");
      }
    });

    window.addEventListener("resize", () => {
      drawLinks();
    });
  }

  function init() {
    if (!loadSaved()) seedData();
    attachEvents();
    renderTaskColors();
    renderAll();
    renderPlanner();
    saveSoon();
  }

  init();
})();
